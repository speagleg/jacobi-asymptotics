\section{Preliminaries}\label{sec:prelim}

This section fixes conventions and collects the inputs of the proof. Two elementary facts about slowly varying
sequences are used repeatedly: they are locally almost constant on dyadic scales, and they grow more slowly than any
power. We prove both in \S\ref{sec:prelim-sv}. The only input from the companion paper~\cite{Paper1} is the explicit
equilibrium measure (\S\ref{sec:prelim-eq}). Every external result used later is stated in \S\ref{sec:tools}, with its
hypotheses, under a label \ref{tool:T1}--\ref{tool:T4} by which the later sections refer to it.

\subsection{Orthogonal polynomials and Jacobi coefficients}\label{sec:prelim-op}
% src: OUTLINE §2 (convention); ROUTE2_alpha §1; PROOF.md §3.5 (orthonormal conventions)

Let $\sigma$ be a positive Borel measure on $[0,\infty)$ with infinite support and finite moments. Its Hankel
determinants are positive, so the monic orthogonal polynomials $P_n$ ($\deg P_n=n$) exist for every $n\ge0$. Put
$h_n:=\int P_n^2\diff\sigma>0$, $P_{-1}:=0$ and $b_0:=0$. The polynomials satisfy the three-term recurrence
\begin{equation}\label{eq:rec-monic}
  xP_n=P_{n+1}+a_nP_n+b_n^2P_{n-1}\qquad(n\ge0),\qquad b_n^2=\frac{h_n}{h_{n-1}}\quad(n\ge1),
\end{equation}
with $a_n\in\R$ and $b_n>0$ for $n\ge1$. We call $a_n$ and $b_n$ the \emph{Jacobi coefficients} of $\sigma$.%
\footnote{Paper~\cite{Paper1} writes $xP_k=P_{k+1}+a_{k+1}P_k+b_k^2P_{k-1}$; its $a_{n+1}$ is our $a_n$, and $b_n$
agrees.}
Multiplying $\sigma$ by a positive constant does not change them.

The orthonormal polynomials $p_n:=P_n/\sqrt{h_n}$ satisfy
\begin{equation}\label{eq:rec-orth}
  xp_n=b_{n+1}p_{n+1}+a_np_n+b_np_{n-1}\qquad(n\ge0),
\end{equation}
as one sees by dividing \eqref{eq:rec-monic} by $\sqrt{h_n}$ and using $b_n^2\sqrt{h_{n-1}/h_n}=b_n$. Since $xp_j$ has
degree $j+1$, orthonormality gives
\begin{equation}\label{eq:coef-int}
  a_n=\int xp_n^2\diff\sigma,\qquad b_n=\int xp_{n-1}p_n\diff\sigma,\qquad \int xp_np_j\diff\sigma=0\quad(j\le n-2).
\end{equation}
In matrix form, \eqref{eq:rec-orth} says that the symmetric tridiagonal matrix $\mathcal J(\sigma)$ with diagonal
$(a_0,a_1,\dots)$ and off-diagonal $(b_1,b_2,\dots)$ is the Jacobi matrix of $\sigma$. The monic form
\eqref{eq:rec-monic} corresponds to the non-symmetric matrix $J=J(\sigma)$ with $J_{n,n}=a_n$, $J_{n,n+1}=1$ and
$J_{n+1,n}=b_{n+1}^2$; this is the form used in \S\ref{sec:shift}.

\emph{Christoffel's extremal property.} For $y\in\R$ and $m\ge1$ let $\mathsf K_m(y):=\sum_{j<m}p_j(y)^2$. If $\deg P<m$
and $P=\sum_{j<m}c_jp_j$, then the Cauchy--Schwarz inequality gives $P(y)^2\le\bigl(\sum_jc_j^2\bigr)\mathsf
K_m(y)=\mathsf K_m(y)\int P^2\diff\sigma$, with equality for $c_j=p_j(y)$. Hence
\begin{equation}\label{eq:christoffel}
  \frac1{\mathsf K_m(y)}=\min\Bigl\{\int P^2\diff\sigma:\ \deg P<m,\ P(y)=1\Bigr\}.
\end{equation}
If $\sigma$ has an atom of mass $w$ at $y$, then every competitor in \eqref{eq:christoffel} has
$\int P^2\diff\sigma\ge wP(y)^2=w$, and therefore
\begin{equation}\label{eq:Kn-le-1}
  w\,\mathsf K_m(y)\le1 .
\end{equation}

\subsection{Slow variation}\label{sec:prelim-sv}
% src: PROOF.md §3.14 Lemma L0 and its remarks

Recall from Definition~\ref{def:sv} that a sequence $\ell:\N\to(0,\infty)$ is slowly varying if, for every $\lambda\ge1$,
\[
  \osc_n(\lambda):=\max\bigl\{|\log\ell(k)-\log\ell(n)|:\ k\in\N,\ n/\lambda\le k\le\lambda n\bigr\}\longrightarrow0
  \qquad(n\to\infty).
\]
Each $\osc_n(\lambda)$ is a maximum of finitely many finite numbers. Put
\[
  \varepsilon_n:=\sup_{m\ge n}\osc_m(2).
\]
Since $\osc_m(2)\to0$, every $\varepsilon_n$ is finite, the sequence $(\varepsilon_n)$ is non-increasing, and
$\varepsilon_n\to0$.

\begin{lemma}[Dyadic chaining and Potter bounds]\label{lem:potter}
Let $\ell$ be slowly varying.
\begin{enumerate}
\item[\textup{(i)}] For all $n\ge1$ and all integers $k\ge n$,
  $|\log\ell(k)-\log\ell(n)|\le\varepsilon_n\bigl(1+\log_2(k/n)\bigr)$.
\item[\textup{(ii)}] For every $\eta>0$ there is $A_\eta<\infty$ such that $e^{-A_\eta}k^{-\eta}\le\ell(k)\le e^{A_\eta}k^{\eta}$ for
  all $k\ge1$.
\end{enumerate}
\end{lemma}

\begin{proof}
(i) Let $j:=\lfloor\log_2(k/n)\rfloor\ge0$, so that $2^jn\le k<2^{j+1}n$. For $0\le i<j$ the integer $2^{i+1}n$ lies in
$[2^in/2,\,2\cdot2^in]$, so $|\log\ell(2^{i+1}n)-\log\ell(2^in)|\le\osc_{2^in}(2)\le\varepsilon_n$. Also $k\in[2^jn,\,2\cdot
2^jn]$, so $|\log\ell(k)-\log\ell(2^jn)|\le\varepsilon_n$. Summing these $j+1$ steps gives
$|\log\ell(k)-\log\ell(n)|\le(j+1)\varepsilon_n\le\varepsilon_n(1+\log_2(k/n))$.

(ii) Fix $\eta>0$ and choose $N$ with $\varepsilon_N\le\eta\log2$. For $k\ge N$, part (i) with $n=N$ gives
\[
  |\log\ell(k)|\le|\log\ell(N)|+\varepsilon_N+\frac{\varepsilon_N}{\log2}\log\frac kN\le|\log\ell(N)|+\varepsilon_N+\eta\log k .
\]
The finitely many $k<N$ are covered by taking
$A_\eta:=\max\bigl(|\log\ell(N)|+\varepsilon_N,\ \max_{k<N}|\log\ell(k)|\bigr)$.
\end{proof}

\begin{remark}[$\lambda=2$ suffices]\label{rem:lambda2}
If $\osc_n(2)\to0$, then $\osc_n(\lambda)\to0$ for every $\lambda\ge1$. Indeed, Lemma~\ref{lem:potter}(i) uses only
$\osc(2)$. For $n\le k\le\lambda n$ it gives $|\log\ell(k)-\log\ell(n)|\le\varepsilon_n(1+\log_2\lambda)$. For $n/\lambda\le
k<n$, apply it with base point $k$: $|\log\ell(n)-\log\ell(k)|\le\varepsilon_k(1+\log_2(n/k))\le\varepsilon_{\lceil
n/\lambda\rceil}(1+\log_2\lambda)$. Both bounds tend to $0$.
\end{remark}

For every $s>1$, Lemma~\ref{lem:potter}(ii) with $\eta<s-1$ gives $\sum_kk^{-s}\ell(k)<\infty$. So
$\sigma_{p,s,\ell}=\sum_kk^{-s}\ell(k)\delta_{k^{-p}}$ is a finite measure on $(0,1]$ with infinitely many atoms, and
\S\ref{sec:prelim-op} applies to it.

\subsection{Regular rates}\label{sec:prelim-rates}
% src: ROUTE2_alpha §4 (regular rates and the regular majorant)

\begin{definition}\label{def:regular}
A sequence $\eta_n>0$ is a \emph{regular rate} if $\eta_n$ is non-increasing, $n\eta_n$ is non-decreasing, and
$\eta_n\to0$.
\end{definition}

The rate $\eta_n=1/n$ is regular. Every regular rate satisfies
\begin{equation}\label{eq:regular-facts}
  \frac1n\le\frac{\eta_n}{\eta_1},\qquad \eta_{n-1}\le2\eta_n\quad(n\ge2),\qquad \sum_n\eta_n=\infty .
\end{equation}
The first holds because $n\eta_n\ge\eta_1$; the second because $(n-1)\eta_{n-1}\le n\eta_n$; the third follows from
the first.

\begin{remark}[Regular majorants]\label{rem:majorant}
Every non-increasing sequence $\varrho_n>0$ with $\varrho_n\to0$ is dominated by a regular rate, namely
$\eta_n:=\max_{m\le n}(m\varrho_m)/n$. Indeed $\eta_n\ge\varrho_n$ and $n\eta_n$ is non-decreasing by construction;
$\eta_{n+1}\le\max\bigl(\eta_n\,n/(n+1),\varrho_{n+1}\bigr)\le\eta_n$ because $\varrho_{n+1}\le\varrho_n\le\eta_n$; and
$\eta_n\to0$, since $m\varrho_m/n\le\varrho_1/\sqrt n$ for $m\le\sqrt n$ and $m\varrho_m/n\le\varrho_{\lceil\sqrt n\rceil}$
for $\sqrt n<m\le n$. So Theorem~\ref{thm:shift} applies to any monotone rate. We use it only with $\eta_n=1/n$.
\end{remark}

\subsection{The equilibrium measure}\label{sec:prelim-eq}
% src: Paper 1 Theorem 1.3(i)-(ii) (= G2 PROOF.md B*(i)-(ii)); RH_px §2 "Input from G2"

The Riemann--Hilbert analysis of Sections~\ref{sec:frame}--\ref{sec:extraction} is built on the equilibrium measure of the constrained problem solved
in~\cite{Paper1}. We use only its explicit form and the equality of its potential on the band; the optimality
statement \cite[Theorem~1.3(iii)]{Paper1} is not used anywhere in this paper.

Let $c=c(p)$ and $I(p)$ be as in \eqref{eq:cp}, let $\gamma_p:=\log c(p)+p$, and let $\lambda_p(x):=\frac1px^{-1-1/p}$ be the density
of the nodes $k^{-p}$ (the image of Lebesgue measure on $(0,\infty)$ under $u\mapsto u^{-p}$). For $V\in[0,1]$ and
$a,b>0$ write $\mathcal I_V(a,b):=B(a,b)^{-1}\int_0^Vt^{a-1}(1-t)^{b-1}\diff t$ for the regularised incomplete beta
function. For $\beta>0$ let $\omega_\beta$ be the arcsine law on $[0,\beta]$, with density $1/(\pi\sqrt{y(\beta-y)})$.

\begin{proposition}[{\cite[Theorem~1.3(i)--(ii)]{Paper1}}]\label{prop:equilibrium}
Let $p>0$ and let $\mu^*:=\int_0^1\omega_{4ct^{-p}}\diff t$, a probability measure on $(0,\infty)$.
\begin{enumerate}
\item[\textup{(i)}] $\mu^*$ has the density
  \[
    \psi^*(x)=\lambda_p(x)\,\mathcal I_{\min(1,\,x/4c)}\bigl(\tfrac1p+\tfrac12,\tfrac12\bigr),\qquad x>0 .
  \]
  In particular $\psi^*=\lambda_p$ on $[4c,\infty)$ (saturation) and $0<\psi^*<\lambda_p$ on $(0,4c)$.
\item[\textup{(ii)}] The function $U^*(x):=\int\log|x-y|\diff\mu^*(y)$ satisfies $U^*(x)=\gamma_p$ for $0<x\le4c$, and
  $U^*(x)\ge\gamma_p$ for all $x>0$.
\item[\textup{(iii)}] (Hard edge.) As $x\to0^+$,
  \[
    \psi^*(x)=C_*x^{-1/2}\bigl(1+O(x)\bigr),\qquad C_*:=\frac{(4c)^{-1/p-1/2}}{p\,a_*B(a_*,\frac12)},\quad
    a_*:=\frac1p+\frac12 .
  \]
\end{enumerate}
\end{proposition}

\begin{proof}
Parts (i) and (ii) are \cite[Theorem~1.3(i)--(ii)]{Paper1}, transported from the variable $u$ to $x=u^{-p}$: there,
$\mu^*_u$ is the image of $\mu^*$ under $x\mapsto x^{-1/p}$ and has density
$\rho^*(u)=\mathcal I_{\min(1,u^{-p}/4c)}(\frac1p+\frac12,\frac12)$, with $\rho^*=1$ exactly on $(0,(4c)^{-1/p}]$ and
$0<\rho^*<1$ on $((4c)^{-1/p},\infty)$. Since $|\diff u/\diff x|=\lambda_p(x)$, the density of $\mu^*$ in $x$ is
$\lambda_p(x)\rho^*(x^{-1/p})$, which is (i). Part (ii) is quoted verbatim, with $\gamma_p=\log c+p$. For (iii), let
$V=x/4c\le1$. Then $\int_0^Vt^{a_*-1}(1-t)^{-1/2}\diff t=\frac{V^{a_*}}{a_*}(1+O(V))$, so
$\psi^*(x)=\frac1px^{-1-1/p}(x/4c)^{1/p+1/2}\bigl(a_*B(a_*,\frac12)\bigr)^{-1}(1+O(x))$, which is (iii).
\end{proof}

The point $x=4c$ is the soft edge, $(0,4c)$ is the band, and $x=0$ is the hard edge, where $\mu^*$ has the
inverse-square-root density of (iii).

\subsection{Tools}\label{sec:tools}
% src: research/G1/RH/CITATIONS.md entries (WLXZ-v1), (WLXZ-v1, additional quotes for RH_px), (WLXZ-v1, Plan 53),
%      (KMVV), (KMVV, additional quotes), (KMVV, angle freedom), (C-D), (DLMF), (DLMF-2), (DLMF-4), (DLMF-5),
%      (DLMF-5.11, Plan 51), (DLMF-5.5/5.11, Plan 54).

This subsection lists every external result used in Sections~\ref{sec:frame}--\ref{sec:extraction} and the appendices. Each is stated with its hypotheses
and the equation numbers of the source, so that the reader can check them against our use. The later sections refer
to these items as \ref{tool:T1}--\ref{tool:T4}. Throughout, the Cauchy transform on an oriented contour $\Sigma$ is
\[
  \Cauchy f(z):=\frac1{2\pi i}\int_\Sigma\frac{f(s)\diff s}{s-z},\qquad z\in\C\setminus\Sigma,
\]
and $\Cauchy_\pm f$ are its non-tangential boundary values from the left ($+$) and the right ($-$) of $\Sigma$, so that
$\Cauchy_+-\Cauchy_-=1$. For a jump matrix $v$ on $\Sigma$ we write $\Cauchy_vf:=\Cauchy_-\bigl(f(v-I)\bigr)$. Matrices
$\sigma_1,\sigma_2,\sigma_3$ are the Pauli matrices, and $\sigma_3=\mathrm{diag}(1,-1)$.

\medskip
\noindent\textbf{T1. The discrete interpolation problem
(Wu--Lin--Xu--Zhao~\cite{WLXZ}).}\settoollabel{T1}{tool:T1}
Wu, Lin, Xu and Zhao~\cite{WLXZ} develop the Riemann--Hilbert method for discrete orthogonal polynomials on infinitely
many nodes with an accumulation point, for the Tricomi--Carlitz polynomials with a positive parameter (their
standing hypothesis: ``$\alpha$ is a positive parameter''). We do not apply any of their theorems to our measures. We
use three things from their paper: the formulation of the problem, the order of the transformations, and the Airy
construction. Our own versions are written out and verified in Sections~\ref{sec:frame} and~\ref{sec:outer}.

\emph{(a) The problem and its solution by the orthogonal polynomials.} Let $\sigma=\sum_{k\ge1}w_k\delta_{x_k}$ with
distinct nodes $x_k\in(0,1]$, $w_k>0$ and $\sum_kw_k<\infty$, and fix $n\ge1$. Following \cite[(2.14) and
Theorem~2.3]{WLXZ}, which they attribute to Fokas, Its and Kitaev, consider the $2\times2$ matrix
\begin{equation}\label{eq:Y-solution}
  Y(z):=\begin{pmatrix}P_n(z) & \displaystyle\sum_k\frac{w_kP_n(x_k)}{z-x_k}\\[3mm]
  h_{n-1}^{-1}P_{n-1}(z) & \displaystyle h_{n-1}^{-1}\sum_k\frac{w_kP_{n-1}(x_k)}{z-x_k}\end{pmatrix}.
\end{equation}
Then: $Y$ is analytic in $\C\setminus(\{x_k\}_k\cup\{0\})$; at each node it has a simple pole with
\[
  \Res_{z=x_k}Y(z)=\lim_{z\to x_k}Y(z)\begin{pmatrix}0&w_k\\0&0\end{pmatrix}
\]
(the residue condition \cite[(2.14)]{WLXZ}); and $Y(z)z^{-n\sigma_3}=I+Y_1/z+Y_2/z^2+O(z^{-3})$ as $z\to\infty$. Moreover
\begin{equation}\label{eq:Y-extract}
  (Y_1)_{12}(Y_1)_{21}=b_n^2,\qquad \frac{(Y_2)_{12}}{(Y_1)_{12}}-(Y_1)_{22}=a_n .
\end{equation}
\emph{Proof.} The sums converge uniformly on compact subsets of $\C\setminus(\{x_k\}\cup\{0\})$, because $|P_j(x_k)|$ is
bounded and $\sum w_k<\infty$; this gives analyticity and the simple poles, and $\Res_{x_k}Y_{j2}=w_kY_{j1}(x_k)$, which is
the residue condition. For $|z|>2$ expand $(z-x)^{-1}=\sum_{i\ge0}x^iz^{-i-1}$, uniformly for $x\in(0,1]$; then
$\sum_kw_kP_m(x_k)/(z-x_k)=\sum_{i\ge0}z^{-i-1}\int P_mx^i\diff\sigma$. Write $P_m(x)=x^m+\pi_mx^{m-1}+\dots$. Then
$\int P_mx^i\diff\sigma=0$ for $i<m$, $=h_m$ for $i=m$, and $=-\pi_{m+1}h_m$ for $i=m+1$ (since
$x^{m+1}=P_{m+1}-\pi_{m+1}P_m+{}$lower terms). Hence $z^nY_{12}=h_nz^{-1}-\pi_{n+1}h_nz^{-2}+O(z^{-3})$,
$z^nY_{22}=1-\pi_nz^{-1}+O(z^{-2})$, $z^{-n}Y_{21}=h_{n-1}^{-1}z^{-1}+O(z^{-2})$ and $z^{-n}Y_{11}=1+O(z^{-1})$. This
gives the normalisation at $\infty$, $(Y_1)_{12}=h_n$, $(Y_1)_{21}=h_{n-1}^{-1}$, $(Y_2)_{12}=-\pi_{n+1}h_n$ and
$(Y_1)_{22}=-\pi_n$. So $(Y_1)_{12}(Y_1)_{21}=h_n/h_{n-1}=b_n^2$, and comparing the coefficients of $x^n$ in
\eqref{eq:rec-monic} gives $\pi_n=\pi_{n+1}+a_n$, i.e.\ the second identity in \eqref{eq:Y-extract}. \qed

No uniqueness or existence statement for this problem is used: the matrix in Section~\ref{sec:frame} is built from \eqref{eq:Y-solution}.

\emph{(b) Transformations.} The sequence of transformations in Section~\ref{sec:frame} follows \cite[\S\S3--5]{WLXZ}: removal of the poles
by the triangular factors built from $\Pi=\sin\theta/\theta'$ and $e^{\pm i\theta}$ (\cite[(3.9)--(3.15)]{WLXZ}),
normalisation by the $g$-function (\cite[(4.1)]{WLXZ}), the lens and saturated-region factors $V_\pm$,
$V^\triangle_\pm$ (\cite[(4.12)--(4.13)]{WLXZ}; in (4.12) the first two cases carry the superscript $\triangle$), and
the conjugation by the outer scalar (\cite[(5.1)]{WLXZ}). In \cite{WLXZ} these are written in a symmetric variable
with nodes $\pm x_k$; we use only their pattern, and every jump is recomputed in Section~\ref{sec:frame} for our one-sided problem.

\emph{(c) The Airy parametrix at a soft edge.} \cite[\S6.1]{WLXZ} construct, in a disc $U(2,\varepsilon)$ about the
soft edge $2$ of their band, a matrix $V_r(z)=E_r(z)\sigma_2\Psi(\tilde f_n(z))\sigma_2e^{n\tilde\varphi\sigma_3}
(\tilde D/\tilde w^{1/2})^{\sigma_3}$ (\cite[(6.12)]{WLXZ}), with $\Psi$ the Airy model problem,
$\tilde f_n(z)=(-\tfrac32n\tilde\varphi(z))^{2/3}$ (\cite[(6.10)]{WLXZ}) and
$\tilde\varphi(z)=-\tfrac23(z-2)^{3/2}+O((z-2)^2)$ (\cite[(6.11)]{WLXZ}). Its hypotheses are: the local jumps have the
form \cite[(6.4)]{WLXZ} (band $\bigl(\begin{smallmatrix}0&w_0/\tilde D^2\\-\tilde D^2/w_0&0\end{smallmatrix}\bigr)$, vertical
segments $\bigl(\begin{smallmatrix}1&\tilde w/(\tilde D^2e^{2n\tilde\varphi})\\0&1\end{smallmatrix}\bigr)$, and
$\bigl(\begin{smallmatrix}1&0\\\tilde D^2e^{2n\tilde\varphi}/\tilde w&1\end{smallmatrix}\bigr)$ on $(2,2+\varepsilon)$),
with $\tilde w$ analytic and non-vanishing in the disc. Its conclusion is the matching condition
\cite[(6.2)]{WLXZ}: $V_{\mathrm{loc}}(z)N^{-1}(z)=I+O(1/n)$ on $\partial U(2,\varepsilon)$, where $N$ is the global
parametrix.

\medskip
\noindent\textbf{T2. The Bessel model problem (Kuijlaars, McLaughlin, Van Assche and Vanlessen~\cite{KMVV}).}%
\settoollabel{T2}{tool:T2}
We use the explicit Bessel model of \cite[\S6]{KMVV} and its algebraic properties; we do not apply their theorems on
orthogonal polynomials. (Their standing hypothesis \cite[(1.1)]{KMVV}, ``$h(x)$ is real analytic and strictly positive
on $[-1,1]$'', concerns their weight and is not needed for the facts below; see Section~\ref{sec:hardedge}.)

\emph{(a) The model.} Let $\alpha>-1$ and let $\Sigma_\Psi$ consist of the rays $\gamma_1:\arg\zeta=2\pi/3$,
$\gamma_2:\arg\zeta=\pi$, $\gamma_3:\arg\zeta=-2\pi/3$, ``oriented as in Figure~3'' of \cite{KMVV}; in that figure
all three rays are oriented towards the origin, with the $+$ side on the left. The function
\[
  \Psi_\alpha(\zeta)=\begin{pmatrix}I_\alpha(2\zeta^{1/2}) & \frac i\pi K_\alpha(2\zeta^{1/2})\\[1mm]
  2\pi i\zeta^{1/2}I_\alpha'(2\zeta^{1/2}) & -2\zeta^{1/2}K_\alpha'(2\zeta^{1/2})\end{pmatrix},\qquad|\arg\zeta|<2\pi/3,
\]
(\cite[(6.23)]{KMVV}), continued to $2\pi/3<\pm\arg\zeta<\pi$ by the Hankel-function formulas \cite[(6.24)--(6.25)]{KMVV},
satisfies the jump relations
\[
  \Psi_+=\Psi_-\begin{pmatrix}1&0\\e^{\alpha\pi i}&1\end{pmatrix}\ \text{on }\gamma_1,\qquad
  \Psi_+=\Psi_-\begin{pmatrix}0&1\\-1&0\end{pmatrix}\ \text{on }\gamma_2,\qquad
  \Psi_+=\Psi_-\begin{pmatrix}1&0\\e^{-\alpha\pi i}&1\end{pmatrix}\ \text{on }\gamma_3
\]
(\cite[(6.16)--(6.18)]{KMVV}) and is a solution of the Bessel model problem (\cite[Theorem~6.3]{KMVV}). As
$\zeta\to\infty$, ``uniformly as $\zeta\to\infty$, $\zeta\in\C\setminus\Sigma_\Psi$'',
\[
  \Psi_\alpha(\zeta)=(2\pi\zeta^{1/2})^{-\sigma_3/2}\frac1{\sqrt2}
  \begin{pmatrix}1+O(\zeta^{-1/2})&i+O(\zeta^{-1/2})\\ i+O(\zeta^{-1/2})&1+O(\zeta^{-1/2})\end{pmatrix}e^{2\zeta^{1/2}\sigma_3}
\]
(\cite[(6.28)]{KMVV}). KMVV obtain (6.28) by inserting the large-argument expansions of $I_\alpha,K_\alpha$ into (6.23)
for $|\arg\zeta|<2\pi/3$, and those of the Hankel functions into (6.24)--(6.25) in the other two sectors. When the
rays are moved by the angle freedom (b) below, the expansions of \ref{tool:T4} (DLMF \S10.40) give the same formula
(6.28) in the enlarged sector; this is checked where it is used. The behaviour of $\Psi_\alpha$ at $\zeta=0$ is derived
in Lemma~\ref{lem:bessel-local} directly from (6.23) and \ref{tool:T4}; we do not use \cite[(6.19)--(6.21)]{KMVV}
beyond the comparison made there.

\emph{(b) Angle freedom.} After \cite[(6.21)]{KMVV}: ``In the above, we have chosen the angle $2\pi/3$ rather
arbitrarily. It could be replaced by any angle in $(0,\pi)$.'' The lens contours in the disc are then defined as
preimages of the rays under $\zeta=n^2f(z)$, which ``does not depend on $n$''. We use rays at
$\pm(\pi-\vartheta_{\mathrm{lip}})$.

\emph{(c) The left endpoint.} Our hard edge has the band to its right, which is the endpoint $-1$ of \cite{KMVV}. There
the conformal map reverses orientation (``if $z<-1$, then $\zeta>0$, while if $z>-1$, then $\zeta<0$''), and ``the
reversal of the directions is taken care of by conjugating with the Pauli matrix $\sigma_3$'':
$\tilde\Psi(\zeta)=\sigma_3\Psi(\zeta;\alpha\to\beta)\sigma_3$ (\cite[(6.51)]{KMVV}), with the prefactor
$\tilde E_n(z)=N(z)\tilde W(z)^{\sigma_3}\frac1{\sqrt2}\bigl(\begin{smallmatrix}1&i\\i&1\end{smallmatrix}\bigr)
\tilde f(z)^{\sigma_3/4}(2\pi n)^{\sigma_3/2}$ (\cite[(6.53)]{KMVV}); compare the right-endpoint formulas
\cite[(6.27), (6.31)]{KMVV} and $\det E_n\equiv1$ (\cite[(6.33)]{KMVV}). The analyticity of the prefactor is proved in
\cite[proof of Proposition~6.5]{KMVV} by a removability argument: ``all entries of $E_n(z)$ have at most a square-root
singularity at $z=1$. Since we already know that $E_n$ is analytic in a punctured neighborhood of $1$, the isolated
singularity at $1$ is removable.'' In Section~\ref{sec:hardedge} this argument is applied only to a prefactor that does not involve the weight.

\noindent\textbf{T3. The comparison argument (Conway--Deift~\cite{CD}).}\settoollabel{T3}{tool:T3}
The following are quoted from \cite[\S4]{CD}. Their Cauchy operator is normalised as ours, as the identity in (b)
requires ($\Cauchy_+-\Cauchy_-=1$).

\emph{(a) Boundedness and the class $\dC$.} ``The operators $C_\Sigma^\pm$ are bounded from $L^2(\Sigma)\to L^2(\Sigma)$ (and
in fact in $L^p(\Sigma)$ for all $1<p<\infty$) if and only if $\Sigma$ is Carleson.'' A pair of functions $A_\pm\in
L^p(\Sigma)$ belongs to $\dC(L^p(\Sigma))$ ``iff $A_\pm=C_\Sigma^\pm f$ for some (unique) $f\in L^p(\Sigma)$''; we write
$A_\pm\in I+\dC(L^p)$ if $A_\pm-I\in\dC(L^p)$.

\emph{(b) Theorem 4.4 of \cite{CD}.} ``Suppose $v_A,v_B\in L^\infty(\Sigma)$ and $A_\pm$ solves the RHP: $A_+(x)=A_-(x)v_A(x)$,
for $x\in\Sigma$, $A_\pm\in I+\dC(L^p(\Sigma))$, and $B_\pm$ the RHP: $B_+(x)=B_-(x)v_B(x)$, for $x\in\Sigma$, $B_\pm\in
I+\dC(L^p(\Sigma))$. Suppose further that $v_B^{-1}\in L^\infty(\Sigma)$ and $B_\pm^{-1}\in I+\dC(L^q(\Sigma))$ where
$1/p+1/q=1$, then if $A,B$ are the analytic extensions of $A_\pm$ and $B_\pm$ respectively,
\[
  AB^{-1}=I+C_\Sigma\bigl(A_-(v_Av_B^{-1}-I)B_-^{-1}\bigr).\text{''}
\]
\emph{Consequence (moments at infinity).} If in addition $\Sigma$ is bounded and $A=I+A_1/z+A_2/z^2+O(z^{-3})$,
$B=I+B_1/z+B_2/z^2+O(z^{-3})$ as $z\to\infty$, then with $H:=A_-(v_A-v_B)v_B^{-1}B_-^{-1}\in L^1(\Sigma)$ (take $p=q=2$),
\begin{equation}\label{eq:CD-moments}
  A_1-B_1=-\frac1{2\pi i}\int_\Sigma H(s)\diff s,\qquad
  A_2-B_2=-\frac1{2\pi i}\int_\Sigma sH(s)\diff s+(A_1-B_1)B_1 .
\end{equation}
Indeed, for $|z|>\max_\Sigma|s|$ we have $\Cauchy H(z)=-\sum_{j\ge0}z^{-j-1}\frac1{2\pi i}\int_\Sigma s^jH(s)\diff s$, while
$AB^{-1}=I+(A_1-B_1)z^{-1}+(A_2-B_2-(A_1-B_1)B_1)z^{-2}+O(z^{-3})$. This is the mechanism of
\cite[Proposition~4.9, (4.14)]{CD}, whose proof identifies ``$\mu=Q_-$, $\tilde\mu=\tilde Q_-$''.

\emph{(c) Solution formula.} \cite[\S4.2]{CD}: with $(C_hf)(x)=C_\Sigma^-(f(h-I))(x)$, ``If $1-C_{v_\Sigma}$ is an invertible
operator from $L^2(\Sigma)$ to $L^2(\Sigma)$, then $\mu=(1-C_{v_\Sigma})^{-1}I$ \dots\ the unique solution $X$ to the RHP
normalized at infinity with jump $v_\Sigma$ is given by $X(z)=I+C_\Sigma(\mu(v_\Sigma-I))$.''

\emph{(d) The Neumann-series step} of \cite[proof of Thm.~4.7, (4.12)--(4.13)]{CD}. If
$\|(1-\Cauchy_{\tilde v})^{-1}\|_{L^2\to L^2}\le K$ and $\|\Cauchy_v-\Cauchy_{\tilde v}\|_{L^2\to L^2}\le1/(2K)$, then
\[
  (1-\Cauchy_v)^{-1}=(1-\Cauchy_{\tilde v})^{-1}\bigl(1-(\Cauchy_v-\Cauchy_{\tilde v})(1-\Cauchy_{\tilde v})^{-1}\bigr)^{-1}
\]
exists and has norm at most $2K$. (This is elementary: the inner factor is $1$ minus an operator of norm at most
$\frac12$.) In \cite{CD} the smallness comes from ``$\|v_\Sigma-\tilde v_\Sigma\|_{L^\infty(\Sigma)}\to0$, which, in particular,
implies that $\|C_{v_\Sigma}-C_{\tilde v_\Sigma}\|_{L^2\to L^2}\to0$''.

\emph{(e) Proposition 4.6 of \cite{CD}} (attributed there to Deift--Zhou): ``A family of operators $(1-C_{h_n})^{-1}$ is
bounded uniformly with respect to $n$ if and only if the inhomogeneous RHP $m_+(s)=m_-(s)h_n(s)+g(s)$ for $s\in\Sigma$,
$m_\pm\in\dC(L^2(\Sigma))$ is (uniquely) solvable for all $g\in L^2(\Sigma)$ with $\|m_\pm\|_{L^2}\le c\|g\|_{L^2}$ with
$c$ independent of both $n$ and $g$.'' We use only the direction ``solvable with bounds $\Rightarrow$ bounded inverse'',
and we re-prove it in Appendix~\ref{app:resolvent}.

We do \emph{not} use \cite[Theorem~4.5]{CD}, whose proof is specific to the Legendre comparison problem.

\noindent\textbf{T4. Special functions (NIST DLMF~\cite{DLMF}).}\settoollabel{T4}{tool:T4}
Principal branches throughout. We use:
\begin{itemize}
\item \emph{Gamma function.} (5.5.1) $\Gamma(z+1)=z\Gamma(z)$; (5.5.3) $\Gamma(z)\Gamma(1-z)=\pi/\sin(\pi z)$ for
  $z\ne0,\pm1,\dots$; (5.11.1) as $z\to\infty$ in $|\mathrm{ph}\,z|\le\pi-\delta$,
  $\mathrm{Ln}\,\Gamma(z)\sim(z-\frac12)\ln z-z+\frac12\ln(2\pi)+\sum_{k\ge1}\frac{B_{2k}}{2k(2k-1)z^{2k-1}}$; and the
  remainder bound of \S5.11(ii): if the sum is terminated at $k=m-1$ and $z$ is complex, the remainder is bounded in
  magnitude by $\sec^{2m}(\frac12\mathrm{ph}\,z)$ times the first neglected term.
\item \emph{Beta integrals.} (5.12.1) $B(a,b)=\int_0^1t^{a-1}(1-t)^{b-1}\diff t=\Gamma(a)\Gamma(b)/\Gamma(a+b)$ and (5.12.3)
  $\int_0^\infty t^{a-1}(1+t)^{-a-b}\diff t=B(a,b)$, for $\Re a,\Re b>0$.
\item \emph{Mellin transform.} (1.14.32) $\mathcal Mf(s)=\int_0^\infty x^{s-1}f(x)\diff x$. (1.14.38) For real $f$,
  \[
    \int_0^\infty f(x)^2\diff x=\frac1{2\pi}\int_{-\infty}^\infty\bigl|\mathcal Mf(\tfrac12+it)\bigr|^2\diff t .
  \]
  (1.14.39)--(1.14.40) The Mellin convolution $(f*g)(x):=\int_0^\infty f(y)g(x/y)\diff y/y$ satisfies
  $\int_0^\infty x^{s-1}(f*g)(x)\diff x=\mathcal Mf(s)\,\mathcal Mg(s)$ when $x^{\Re s-1}f$ and $x^{\Re s-1}g$
  are absolutely integrable.
\item \emph{Modified Bessel functions.} (10.25.2)
  $I_\nu(z)=(\frac12z)^\nu\sum_{k\ge0}(\frac14z^2)^k/(k!\,\Gamma(\nu+k+1))$, the principal branch being analytic in
  $\C\setminus(-\infty,0]$ (\S10.25(ii)); (10.27.3) $K_{-\nu}=K_\nu$; (10.27.4)
  $K_\nu(z)=\frac12\pi(I_{-\nu}(z)-I_\nu(z))/\sin(\nu\pi)$; (10.31.2)
  $K_0(z)=-(\ln(\frac12z)+\gamma)I_0(z)+\frac{\frac14z^2}{(1!)^2}+(1+\frac12)\frac{(\frac14z^2)^2}{(2!)^2}+\dotsb$ ($\gamma$
  Euler's constant).
\item \emph{Wronskian.} (1.13.4) $\mathcal W\{w_1,w_2\}=w_1w_2'-w_2w_1'$ and (10.28.2)
  $\mathcal W\{K_\nu(z),I_\nu(z)\}=I_\nu(z)K_{\nu+1}(z)+I_{\nu+1}(z)K_\nu(z)=1/z$. Together with (6.23) of \ref{tool:T2}
  they give $\det\Psi_\alpha=2\zeta^{1/2}\mathcal W\{K_\alpha,I_\alpha\}(2\zeta^{1/2})=1$ for $|\arg\zeta|<2\pi/3$.
\item \emph{Large argument.} As $z\to\infty$ with $\nu$ fixed (\S10.40(i)),
  \begin{align*}
    &(10.40.1)&I_\nu(z)&\sim\frac{e^z}{(2\pi z)^{1/2}}\sum_{k\ge0}(-1)^k\frac{a_k(\nu)}{z^k},&&|\mathrm{ph}\,z|\le\tfrac12\pi-\delta,\\
    &(10.40.2)&K_\nu(z)&\sim\Bigl(\frac\pi{2z}\Bigr)^{1/2}e^{-z}\sum_{k\ge0}\frac{a_k(\nu)}{z^k},&&|\mathrm{ph}\,z|\le\tfrac32\pi-\delta,
  \end{align*}
  and (10.40.3)--(10.40.4) are the same expansions for $I_\nu'$, $K_\nu'$ with $a_k(\nu)$ replaced by $b_k(\nu)$ and an
  overall minus sign for $K_\nu'$, in the same sectors. The ``more general (and more accurate)'' (10.40.5) is
  \[
    I_\nu(z)\sim\frac{e^z}{(2\pi z)^{1/2}}\sum_{k\ge0}(-1)^k\frac{a_k(\nu)}{z^k}\pm ie^{\pm\nu\pi i}\frac{e^{-z}}{(2\pi
    z)^{1/2}}\sum_{k\ge0}\frac{a_k(\nu)}{z^k},\qquad -\tfrac12\pi+\delta\le\pm\mathrm{ph}\,z\le\tfrac32\pi-\delta .
  \]
  Here $\delta>0$ is a small constant, and $a_0(\nu)=1$ by (10.17.1) (an empty product).
\end{itemize}
