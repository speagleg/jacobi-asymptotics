\title[Recurrence coefficients with slowly varying weights]{Recurrence coefficients of discrete measures with
power-law atoms and slowly varying weights}
\author{Gordon [surname] (draft)}
\date{Draft of 14 October 2026}

\begin{abstract}
Let $\sigma=\sum_{k\ge1}k^{-s}\ell(k)\,\delta_{k^{-p}}$ with $p>0$, $s>1$ and $\ell$ slowly varying; its Jacobi
matrix is a compact operator with eigenvalues $k^{-p}$. We prove that the recurrence coefficients satisfy
$n^pb_n\to c(p)$ and $n^pa_n\to2c(p)$, where
$c(p)=\frac14\bigl(\sqrt\pi\,\Gamma(1/p+1)/\Gamma(1/p+\frac12)\bigr)^p$ is the constant of the constrained
equilibrium problem solved in a companion paper. This holds for every slowly varying $\ell$, including unbounded ones
such as $\exp\sqrt{\log k}$. If $\ell$ extends to a slowly varying analytic function on a sector, the error is $O(1/n)$; for pure powers this
gives $n^pb_n=c(p)(1+O(1/n))$ and $a_n/(2b_n)=1+O(1/n)$. The proof reduces, by a Christoffel transformation (which
shifts $s$ by $p$ and preserves the law with its rate) and a smoothing step, to analytic factors and the window
$1<s<1+2p$. There a Riemann--Hilbert analysis of discrete orthogonal polynomials whose nodes accumulate at a hard edge
controls the non-analytic factor by a Conway--Deift comparison with a Bessel model, instead of a local parametrix. In
a further paper we show that bounded but rough weights can destroy the pointwise law.
\end{abstract}

\section{Introduction}\label{sec:intro}

\subsection{Setting}\label{sec:setting}
A compact Jacobi operator is determined by its spectral measure. If its eigenvalues are $k^{-p}$ and its spectral
weights decay like $k^{-s}$, how do its matrix entries behave? We show that the off-diagonal entries $b_n$ and the
diagonal entries $a_n$ satisfy $n^pb_n\to c(p)$ and $n^pa_n\to2c(p)$, with an explicit constant $c(p)$, for every
$p>0$, every $s>1$ and every slowly varying perturbation of the weights, and that the convergence has rate $O(1/n)$
when the perturbation is analytic.

Concretely, for $p>0$, $s>1$ and a sequence $\ell\colon\N\to(0,\infty)$ we study the discrete measure
\begin{equation}\label{eq:sigma}
  \sigma_{p,s,\ell}:=\sum_{k\ge1}k^{-s}\ell(k)\,\delta_{k^{-p}},\qquad \sigma_{p,s}:=\sigma_{p,s,1}.
\end{equation}
It lives on $(0,1]$, and its infinitely many atoms accumulate only at $0$. Let $P_n$ be its monic orthogonal
polynomials and $h_n:=\int P_n^2\diff\sigma$. They satisfy the three-term recurrence
\begin{equation}\label{eq:jacobi}
  xP_n=P_{n+1}+a_nP_n+b_n^2P_{n-1}\qquad(n\ge0,\ b_0:=0,\ b_n>0),
\end{equation}
so that $b_n^2=h_n/h_{n-1}$; the numbers $a_n$ and $b_n$ are the entries of the Jacobi matrix of $\sigma$. With
\begin{equation}\label{eq:cp}
  I(p):=\frac{\sqrt\pi}{2}\,\frac{\Gamma(\frac1p+\frac12)}{\Gamma(\frac1p+1)},\qquad
  c(p):=\frac14\Bigl(\frac{\pi}{2I(p)}\Bigr)^{p}
  =\frac14\Bigl(\frac{\sqrt\pi\,\Gamma(\frac1p+1)}{\Gamma(\frac1p+\frac12)}\Bigr)^{p},
\end{equation}
for example $c(1)=\frac12$ and $c(2)=\pi^2/16$, the question is whether
\begin{equation}\label{eq:law}
  n^pb_n\longrightarrow c(p)\qquad\text{and}\qquad n^pa_n\longrightarrow2c(p)\qquad(n\to\infty).
\end{equation}
We call \eqref{eq:law} the \emph{pointwise law}. The constant $c(p)$ comes from the companion paper \cite{Paper1},
which proves \eqref{eq:law} on average (see \eqref{eq:paper1-averaged} below).

\subsection{Results}\label{sec:results}
We use the following form of slow variation, which needs no measurability or representation theorem.

\begin{definition}[slow variation]\label{def:sv}
A sequence $\ell\colon\N\to(0,\infty)$ is \emph{slowly varying} if, for every $\lambda\ge1$,
\begin{equation}\label{eq:osc}
  \osc_n(\lambda):=\max\bigl\{\,|\log\ell(k)-\log\ell(n)|:\ k\in\N,\ n/\lambda\le k\le\lambda n\,\bigr\}
  \longrightarrow0\qquad(n\to\infty).
\end{equation}
\end{definition}

The case $\lambda=2$ already implies the others (Remark~\ref{rem:lambda2}). Examples are $(\log(k+1))^\beta$ with
$\beta\in\R$, $\exp((\log k)^\beta)$ with $0<\beta<1$, and $\exp(\sqrt{\log k}\,\sin\log\log k)$ for $k\ge3$. Neither
boundedness nor monotonicity is required.

For the sharp rate we need an analytic extension of $\ell$ to a sector, written in the logarithmic variable
$\xi=\log t$. For $\vartheta>0$ and $c_0>0$ put $S_\vartheta:=\{\xi\in\C:\ \Re\xi\ge-c_0,\ |\Im\xi|\le\vartheta\}$.

\begin{definition}[analytic slowly varying factors]\label{def:A}
Let $\vartheta>0$. A sequence $\ell$ satisfies \Ath{\vartheta} if $\ell(k)=\exp\Lambda(\log k)$ for all $k\ge1$,
where, for some $c_0>0$, the function $\Lambda$ is analytic on $S_\vartheta$, real on $S_\vartheta\cap\R$, bounded on
$S_\vartheta\cap\{\Re\xi\le0\}$, and
\begin{equation}\label{eq:epsLambda}
  \varepsilon_\Lambda(T):=\sup\bigl\{|\Lambda'(\xi)|:\ \xi\in S_\vartheta,\ \Re\xi\ge T\bigr\}\longrightarrow0
  \qquad(T\to\infty).
\end{equation}
We then also say that $\Lambda$ satisfies \Ath{\vartheta}. We use the half-width
\begin{equation}\label{eq:thetap}
  \vartheta_p:=\pi/p\quad\text{for }p>1,\qquad\vartheta_p:=\pi/2\quad\text{for }0<p\le1.
\end{equation}
\end{definition}

\begin{maintheorem}[main theorem]\label{thm:main}
Let $p>0$, $s>1$, let $\ell\colon\N\to(0,\infty)$ be slowly varying, and let $\sigma=\sigma_{p,s,\ell}$. Then
\begin{enumerate}
\item[\textup{(i)}] $n^pb_n\to c(p)$ and $n^pa_n\to2c(p)$;
\item[\textup{(ii)}] if $\ell$ satisfies \Ath{\vartheta_p}, then $n^pb_n=c(p)(1+O(1/n))$ and
  $a_n/(2b_n)=1+O(1/n)$.
\end{enumerate}
\end{maintheorem}

In words, \Ath{\vartheta_p} asks that $\ell$ extend to an analytic function $\ell(t)=e^{\Lambda(\log t)}$
on the set $\{|t|\ge e^{-c_0},\ |\arg t|\le\vartheta_p\}$, bounded near $|t|=e^{-c_0}$, with $t\ell'(t)/\ell(t)\to0$. Such an
$\ell$ is slowly varying, since $|\Lambda(\log k)-\Lambda(\log n)|\le\varepsilon_\Lambda(\log
n-\log\lambda)\log\lambda$ for $n/\lambda\le k\le\lambda n$. The sequence $\ell(k)=\exp\sqrt{\log(k+e)}$ satisfies
\Ath{\vartheta_p} for every $p>0$, because the branch points of $\Lambda(\xi)=\sqrt{\log(e^\xi+e)}$ lie on the lines
$|\Im\xi|=\pi$. For $p>1$ a further example is $\Lambda(\xi)=(\log(e^\xi+e))^\beta$ with $\beta<1$.

\emph{Why the pointwise law is subtle.} Corollary~1.2 of \cite{Paper1}, specialised to the atoms $x_k=k^{-p}$,
states: if the weights $w_k\ge0$ satisfy $\sum_kw_k<\infty$ and $w_k\ge c_1k^{-s'}$ for some constants $c_1>0$ and
$s'\in\R$, then the recurrence coefficients of $\sum_kw_k\delta_{k^{-p}}$ satisfy
\begin{equation}\label{eq:paper1-averaged}
  \frac{2}{n^2}\sum_{j=1}^{n-1}(n-j)\log\bigl(j^pb_j\bigr)\longrightarrow\log c(p).
\end{equation}
The constant comes from an explicitly solvable constrained equilibrium problem \cite[Theorem~1.3]{Paper1}. Its
solution $\mu^*$ is an average of arcsine laws; it is the zero distribution that a Jacobi matrix with entries
$a_n\approx2b_n\approx2c\,n^{-p}$ would produce, and the saturation condition of the problem fixes $c=c(p)$
\cite[\S1.2]{Paper1}. So $c(p)$ is the value predicted for the pointwise law. But \eqref{eq:paper1-averaged} is a
double Ces\`aro mean of $\log(j^pb_j)$, that is, a statement about the logarithm of the Hankel determinant $D_n$ of
$\sigma$ at order $n^2$. Multiplying the weights by factors in $[e^{-\varepsilon},e^{\varepsilon}]$ changes $\log D_n$ by
at most $\varepsilon n$, so the averaged law cannot see such perturbations, and passing from it to \eqref{eq:law} is a
Tauberian step that needs a regularity of $j^pb_j$ which the averaged argument does not provide. That regularity must come from the weights: the
averaged law holds under the one-sided bound $w_k\ge c_1k^{-s'}$ alone, while the pointwise law can fail for weights
that oscillate within a fixed factor of $k^{-s}$ (\S\ref{sec:context}). Theorem~\ref{thm:main} shows that slow
variation of the weights is enough.

\begin{definition}[smoothing]\label{def:smoothing}
Let $L_\ell\colon\R\to\R$ be the piecewise-linear interpolant of the points $(\log k,\log\ell(k))$, $k\ge1$, extended
by the constant $\log\ell(1)$ on $(-\infty,0]$. Let $G(y):=e^{-y^2/2}/\sqrt{2\pi}$ and
\begin{equation}\label{eq:smoothing}
  \Lambda_\ell:=L_\ell*G,\qquad
  \rho_\ell(m):=\sup_{k\ge m}\bigl|\log\ell(k)-\Lambda_\ell(\log k)\bigr|\qquad(m\ge1).
\end{equation}
\end{definition}

\emph{Quantitative form.} The proof of Theorem~\ref{thm:main}(i) gives a rate, measured by how well $\ell$ can be
smoothed in the logarithmic variable. The smoothing error $\rho_\ell(m)$ of Definition~\ref{def:smoothing} tends to
$0$ for every slowly varying $\ell$ (Lemma~\ref{lem:smoothing}), and
\begin{equation}\label{eq:main-rate}
  |n^pb_n-c(p)|+|n^pa_n-2c(p)|=O\bigl(1/n+\rho_\ell(\ceil{n/M_0})\bigr),
\end{equation}
where the head margin $M_0=M_0(p)\ge4$ is an integer that depends only on $p$ (Lemma~\ref{lem:saturation}). The
indices $k<n/M_0$ are those at which the spectral weights of $P_n$ are exponentially saturated.


\begin{remark}\label{rem:main-finite}
No boundedness of $\ell$ is assumed. For example, the weights $k^{-s}\exp\sqrt{\log k}$ and
$k^{-s}\exp(\sqrt{\log k}\,\sin\log\log k)$ ($k\ge3$) are covered; the second is not even eventually comparable with
any fixed power of $\log k$. The measure is always finite, by the Potter bounds (\S\ref{sec:prelim-sv}), so its Jacobi
matrix is a compact operator with eigenvalues $k^{-p}$.
\end{remark}

\begin{remark}[what is, and is not, proved at order $1/n$]\label{rem:main-rate}\label{rem:engine-scope}
Part (ii) bounds the deviation from $c(p)$ at order $1/n$ but does not identify the $1/n$ term. The proof compares the
true problem with an exactly solvable comparison problem: relative to it, the non-analytic part of $\ell$ moves the
first two coefficients at infinity by $O(\varepsilon'_n/n)$ with $\varepsilon'_n\to0$ (Section~\ref{sec:hardedge}),
but the comparison problem itself depends on $\ell$ at order $1/n$. So the $O(1/n)$ is absolute, not relative: we do
not claim that $b_n(\sigma_{p,s,\ell})/b_n(\sigma_{p,s})-1=O(\varepsilon_\Lambda/n)$. A conjecture for the $1/n$ term,
supported by numerics, is formulated in \S\ref{sec:tilt}.
\end{remark}

For pure powers, $\ell\equiv1$ satisfies \Ath{\vartheta} for every $\vartheta$, and Theorem~\ref{thm:main}(ii) gives
the following.

\begin{corollary}[pure powers]\label{thm:pure}
For every $p>0$ and $s>1$ the measure $\sigma_{p,s}=\sum_{k\ge1}k^{-s}\delta_{k^{-p}}$ satisfies
$n^pb_n=c(p)(1+O(1/n))$ and $a_n/(2b_n)=1+O(1/n)$. For example, $nb_n=\frac12+O(1/n)$ for $\sigma_{1,s}$ and
$n^2b_n=\pi^2/16+O(1/n)$ for $\sigma_{2,s}$.
\end{corollary}

\begin{remark}[a classical instance]\label{rem:pure-lommel}
For $(p,s)=(2,2)$ the coefficients are explicit. Van Assche \cite[\S3]{VA} records that the orthonormal Lommel
polynomials satisfy $xp_n=\beta_{n+1}p_{n+1}+\beta_np_{n-1}$ with $\beta_{n+1}=[2\sqrt{(n+\nu)(n+\nu+1)}]^{-1}$. For
$\nu=\frac32$ the orthogonality measure $\mu$ is identified as follows. By the Bessel recurrence
\cite[(10.6.1)]{DLMF}, $u_n:=\sqrt{n+\nu}\,J_{\nu+n}(1/x)$ solves $xu_n=\beta_{n+1}u_{n+1}+\beta_nu_{n-1}$ for $n\ge0$
(with $\beta_0:=[2\sqrt{\nu(\nu-1)}]^{-1}$), and it is square summable by \cite[(10.2.2)]{DLMF}. For $x\notin\R$ the
resolvent vector $(J-x)^{-1}e_0$ of the Jacobi matrix $J$ solves the same recurrence for $n\ge1$ and is square
summable, and such solutions form a line, since the Wronskian $\beta_n(u_nv_{n-1}-u_{n-1}v_n)$ of two of them is
constant and tends to $0$. Hence $\int(t-x)^{-1}\diff\mu(t)$, the first entry of the resolvent, equals $-u_0/(\beta_0u_{-1})
=-2\nu J_\nu(1/x)/J_{\nu-1}(1/x)$. For $\nu=\frac32$ this is $-3(z^{-1}-\cot z)$ with $z=1/x$
\cite[(10.16.1), (10.6.1)]{DLMF}, and the partial fractions of $\cot$ \cite[(4.22.3)]{DLMF} show that $\mu$ has mass
$3/(\pi k)^2$ at $\pm1/(\pi k)$, $k\ge1$. Its image under $t\mapsto\pi^2t^2$ is $(6/\pi^2)\sigma_{2,2}$, and the
quadratic transformation (from $P_{2n}(t)=Q_n(t^2)$, where $Q_n$ are the monic orthogonal polynomials of the image: $a_n=\pi^2(\beta_{2n}^2+\beta_{2n+1}^2)$, $b_n=\pi^2\beta_{2n-1}\beta_{2n}$) gives
\[
  b_n=\frac{\pi^2}{(4n+1)\sqrt{(4n-1)(4n+3)}},\qquad a_n=\frac{2\pi^2}{(4n+1)(4n+5)}\qquad(n\ge1)
\]
for $\sigma_{2,2}$. Thus $n^2b_n/c(2)=1-\frac1{2n}+O(n^{-2})$ and $a_n/(2b_n)=1-\frac1n+O(n^{-2})$, in agreement with
Corollary~\ref{thm:pure}.
\end{remark}

\begin{remark}[Hankel determinants of zeta values]\label{rem:pure-zeta}
For $a>0$ and $2a+b>1$ the measure $\sigma_{a,2a+b}$ has moments $\int x^j\diff\sigma_{a,2a+b}=\zeta(aj+2a+b)$, so
its Hankel determinants are $\det_{1\le i,j\le n}\zeta(a(i+j)+b)$ \cite[Corollary~1.4]{Paper1}.
Corollary~\ref{thm:pure} gives $n^ab_n=c(a)(1+O(1/n))$ for the recurrence coefficients behind these determinants. For
$a=1$ Monien \cite{Monien} reports a full asymptotic expansion of the determinants, obtained numerically by
D.~Zagier; it goes beyond Corollary~\ref{thm:pure} but is not proved.
\end{remark}

\begin{remark}[where $c(p)$ comes from]\label{rem:pure-constant}
The equilibrium measure $\mu^*$ of \cite{Paper1} is supported on $(0,\infty)$ and coincides with the node density on
$[4c(p),\infty)$ (it is \emph{saturated} there); its unconstrained part is the band $(0,4c(p))$
(Proposition~\ref{prop:equilibrium}). After rescaling by $n^p$, the Riemann--Hilbert analysis sees a band of length
$4c(p)$, and a band of length $\beta$ produces off-diagonal entries close to $\beta/4$.
\end{remark}

The second result is elementary, and it is the reason why the analytic part of the proof only has to cover a window
of exponents $s$. A sequence $\eta_n>0$ is a \emph{regular rate} if $\eta_n$ is non-increasing, $n\eta_n$ is
non-decreasing, and $\eta_n\to0$; for example $\eta_n=1/n$ (see \S\ref{sec:prelim-rates}).

\begin{maintheorem}[Christoffel shift: law and rate]\label{thm:shift}
Let $\sigma$ be a positive Borel measure on $[0,\infty)$ with infinite support and finite moments, let $p>0$ and
$c>0$, and let $\eta_n$ be a regular rate. If
\[
  |n^pa_n(\sigma)-2c|+|n^pb_n(\sigma)-c|=O(\eta_n),
\]
then the measure $x\sigma$ \textup{(}$\diff(x\sigma):=x\diff\sigma(x)$\textup{)} satisfies the same estimate, with
the same $c$ and rate $O(\eta_n)$. Without a rate: if $n^pa_n(\sigma)\to2c$ and $n^pb_n(\sigma)\to c$, then the same
holds for $x\sigma$. For $x_k=k^{-p}$ one has exactly $x\cdot\sigma_{p,s,\ell}=\sigma_{p,s+p,\ell}$.
\end{maintheorem}

\begin{remark}\label{rem:shift-credit}
The Jacobi matrix of $x\sigma$ is obtained from that of $\sigma$ by factorising $J=LU$ and reversing the factors; these
transfer formulas are classical \cite[\S1]{DD}. The factors are governed by a chain sequence, and the statement
without rate follows from a proposition of Dai, Piao and Qiao \cite[Proposition~2.1]{DPQ} together with those formulas.
The part with a rate is the one that matters here: it lets the rate $O(1/n)$ survive the shift. The parameter
sequence of the chain converges only at the rate $O(\sqrt{\eta_n})$ in general, and this is sharp; the coefficients of
$x\sigma$ involve only its increments, which are $O(\eta_n)$ (Section~\ref{sec:shift}). We have not found the rate
statement in the literature.
\end{remark}

\begin{remark}\label{rem:shift-converse}
Conversely, without rate: if $x\sigma$ satisfies \eqref{eq:law} (with any $c>0$ in place of $c(p)$), so does
$\sigma$. In particular the pointwise law is insensitive to a point mass at $0$ (Remark~\ref{rem:converse}).
\end{remark}

\subsection{Strategy of the proof}\label{sec:strategy}
The proof has a soft part and a hard part. The soft part (Sections~\ref{sec:shift} and~\ref{sec:smoothing}) is
elementary and self-contained; it reduces Theorem~\ref{thm:main} to the following statement, which is the technical
core of the paper.

\begin{maintheorem}[the Riemann--Hilbert engine: analytic factor, base window]\label{thm:engine}
Let $p>0$ and $1<s<1+2p$, i.e.\ $\alpha:=(s-1-p)/p\in(-1,1)$. Let $\ell_1$ satisfy \Ath{\vartheta_p}. Then
$\sigma_{p,s,\ell_1}$ satisfies
\[
  n^pb_n=c(p)\bigl(1+O(1/n)\bigr),\qquad a_n/(2b_n)=1+O(1/n).
\]
\end{maintheorem}

\emph{From Theorem~\ref{thm:engine} to Theorem~\ref{thm:main}.} Write $s=s_0+mp$ with $s_0\in(1,1+2p)$ and an
integer $m\ge0$; this is possible because the window $(1,1+2p)$ has length $2p>p$. Theorem~\ref{thm:engine} at $s_0$
and $m$ applications of Theorem~\ref{thm:shift} with $\eta_n=1/n$ give the law with rate $O(1/n)$ for every analytic
factor and every $s>1$. This is Theorem~\ref{thm:main}(ii). For a general slowly varying $\ell$ we apply this to the
analytic factor $\ell_1:=\exp\Lambda_\ell(\log\cdot)$ and compare: a multiplicative perturbation $\ell/\ell_1\to1$ of
the weights changes $\log b_n^2$ and $n^pa_n$ by at most a constant times the supremum of $|\log(\ell/\ell_1)|$ over
the indices $k\ge n/M_0$, where the spectral weight of $P_n$ lives, up to an exponentially small contribution from the
head $k<n/M_0$ (Proposition~\ref{prop:stability}). This gives \eqref{eq:main-rate}, and Theorem~\ref{thm:main}(i)
follows.

\emph{Why the window $|\alpha|<1$.} Near the accumulation point $0$ the measure looks like a continuous weight: the
nodes $x_k=k^{-p}$ have density $\frac1px^{-1-1/p}$, so the mass per unit length near $x=k^{-p}$ is
$k^{-s}\cdot\frac1px^{-1-1/p}=\frac1px^{\alpha}$. In the Riemann--Hilbert (RH) analysis $0$ becomes a hard edge with Bessel order
$\alpha=(s-1-p)/p$. Our comparison argument works for $|\alpha|<1$ (Remark~\ref{rem:alpha-ge-1}). Multiplication by
$x$ raises $\alpha$ by $1$, and Theorem~\ref{thm:shift} carries the law and its rate across. For $s>1+p/2$ one can
even choose $s_0$ with $|\alpha|\le\frac12$; the case $\alpha=0$ cannot be avoided in general
(Remark~\ref{rem:base-window}).

\emph{The frame.} After rescaling $z\mapsto n^pz$ the nodes $(n/k)^p$ fill $(0,n^p]$. The equilibrium measure $\mu^*$
of \cite{Paper1} governs the $n$-th root behaviour: on the band $(0,4c(p))$ the polynomials oscillate, on the saturated
region $(4c(p),n^p)$ the zero distribution coincides with the node density, and $4c(p)$ is a soft edge with an
Airy parametrix. We set up the discrete interpolation problem and its first transformations as Wu, Lin, Xu and Zhao
do \cite{WLXZ}, in the one-sided variable $x$ (Section~\ref{sec:frame}). What depends on $p$ is the outer scalar
function that removes the Gamma-function growth of the node masses: the index function $t(z)=nz^{-1/p}$ maps the slit
plane onto the sector $|\arg t|<\pi/p$, which lies inside, exactly fills, or exceeds the sector $|\arg t|<\pi$ where
Stirling's formula holds, according as $p>1$, $p=1$ or $p<1$. Section~\ref{sec:outer} treats the three cases and ends
with a Frame Proposition (Proposition~\ref{prop:frame}), the only interface between the frame and the hard edge.

\emph{The hard edge.} The slowly varying factor enters the band weight through the function
$r_n(z)=\ell_1(nz^{-1/p})/\ell_1(n)$, which is not analytic at $0$. Hence the standard Bessel parametrix, which needs the weight continued around $0$, leaves
a jump of size $O(n\varepsilon_\Lambda)$ at the Bessel scale: a quotient argument fails. Following Conway and Deift
\cite{CD}, we instead build the Bessel parametrix from the Szeg\H{o} function of the whole band weight. The comparison
problem is then exactly solvable, and it differs from the true problem only on the lips of the lens, by an amount that
is $o(1)$ in the supremum norm but has no rate. A Neumann series closes once the comparison resolvent is bounded
uniformly in $n$ (Proposition~\ref{prop:UR-comparison}; this is where $|\alpha|<1$ is used), and Bessel localisation
turns the $o(1)$ mismatch into an $O(1/n)$ error in the coefficients (Section~\ref{sec:hardedge}).

\subsection{Context}\label{sec:context}
\emph{Discrete orthogonal polynomials.} The RH approach to discrete orthogonal polynomials was developed by
Baik, Kriecherbauer, McLaughlin and Miller \cite{BKMM} for measures on finite
node sets in a bounded interval, with real-analytic node density and potential; their analysis includes saturated
regions, which also appear here. Within the literature we have read, the only RH analysis of discrete orthogonal
polynomials on infinitely many nodes accumulating at a finite point is that of Wu, Lin, Xu and Zhao \cite{WLXZ}, for
the Tricomi--Carlitz polynomials; our frame follows theirs. Dai and Yao make the same observation: ``Wu et al.\ [21]
studied the case where the orthogonality lattice consists of infinite nodes with an accumulation point. Except [21],
the orthogonality lattices are always equally spaced'' \cite[\S1]{DY}, [21] being \cite{WLXZ}.

\emph{Non-analytic weights.} Our weights are not analytic at the hard edge, and the standard local parametrix does not
exist there. Conway and Deift \cite{CD} met the same difficulty for the weight $\log\frac{2k}{1-x}$ on $(-1,1)$, which
carries a slowly varying factor at the endpoint $x=1$; they write that ``there is no known parametrix for the
Riemann--Hilbert problem in a neighborhood of the logarithmic singularity at $x=1$'', and they compare the problem with
the Legendre problem through the operator theory behind RH problems instead of through a quotient. The comparison
argument in Section~\ref{sec:hardedge} is that of Conway and Deift \cite[Theorem~4.7 and Proposition~4.9]{CD}. What
we add is a comparison model built from the Szeg\H{o} function of the full weight (an explicit parametrix rather than a
second orthogonality problem), a uniform resolvent bound for a Bessel edge of any order $|\alpha|<1$ (rather than the
Legendre value $\alpha=0$), and the treatment of an arbitrary analytic slowly varying factor (rather than one explicit
logarithm), for a discrete problem whose nodes accumulate at the hard edge. Deift and
Piorkowski \cite{DP} extended the analysis of \cite{CD} to the weight $\log\frac{2}{1-x}$ and write, about a comparison
with a weight that has a simple zero at the endpoint: ``the crucial Theorem~4.7 in [4] requires a different proof in
this case: now one needs to control the behaviour of the Cauchy operator acting on spaces with Muckenhoupt weights''
\cite[\S1]{DP}, [4] being \cite{CD}. This is consistent with our restriction to Bessel orders $|\alpha|<1$
(Remark~\ref{rem:alpha-ge-1}). Other approaches to non-analytic weights are Piorkowski's parametrix-free method, which
needs an a priori estimate for the exact solution \cite{Pio}; the $\bar\partial$ methods of McLaughlin and Miller
\cite{MM} and of Trogdon \cite{Tro}, which assume a finite number of Lipschitz derivatives of the weight (for
\cite{Tro}, $C^{k,1}$ regularity up to the endpoints of the support); and the approximation of a smooth symbol by
analytic ones in Deift, Its and Krasovsky \cite[\S6.2]{DIK}. Our reduction from arbitrary to analytic slowly varying
factors (Section~\ref{sec:smoothing}) is of the last kind, but the approximation error is controlled outside the RH
analysis.

\emph{Recurrence asymptotics without analyticity.} For exponential weights $W=e^{-Q}$, Levin and Lubinsky
\cite{LL} prove $o(1)$ asymptotics of the recurrence coefficients, with constants given by equilibrium quantities, under
regularity conditions that do not involve analyticity, and without RH methods. Theorem~\ref{thm:main} is an analogue
for compact Jacobi operators. We have not found Theorem~\ref{thm:main} in the literature, even for bounded $\ell$. The
closest results we know concern continuous weights on an interval \cite{CD,DP,Pio,Tro,MM} or exponential weights
\cite{LL}.

\emph{Special cases.} Besides the Lommel case (Remark~\ref{rem:pure-lommel}) and the Hankel determinants of zeta
values studied by Monien \cite{Monien} and by Haynes and Zudilin \cite{HaynesZudilin} (Remark~\ref{rem:pure-zeta}), we
mention the Tricomi--Carlitz polynomials. Their orthogonality
measure is supported on $\{\pm1/\sqrt{k+\alpha}\}$ \cite[\S3]{VA}; in the squared
variable its atoms are $1/(k+\alpha)$, so it is an instance of $p=1$ with weights that are not pure powers. For it the
recurrence coefficients are explicit \cite[\S3]{VA}, and \cite{WLXZ} obtain uniform asymptotics of the polynomials.

\emph{Rough weights.} Slow variation cannot be replaced by boundedness. In a further paper \cite{Paper3} we show the
following. For every $p>0$ and $s>1$ there is $\varepsilon_*>0$ such that, for every $\varepsilon\in(0,\varepsilon_*]$,
there are weights $w_k=k^{-s}e^{\epsilon_k}$ with $|\epsilon_k|\le\varepsilon$ for which $n^pb_n$ does not converge,
and weights of the same kind for which $n^pa_n$ does not converge. The averaged law \eqref{eq:paper1-averaged} holds
for all of them. Together with Theorem~\ref{thm:main} this gives a dichotomy: the pointwise law is stable under slowly
varying perturbations of the weights, of any size, and unstable under rough ones, however small.

\subsection{Organisation}\label{sec:organisation}
Section~\ref{sec:prelim} fixes conventions, proves two facts about slowly varying sequences, quotes the input from
\cite{Paper1} (only parts (i) and (ii) of its Theorem~1.3 are used), and states the external tools with their
hypotheses. Section~\ref{sec:shift} proves Theorem~\ref{thm:shift}. Section~\ref{sec:smoothing} reduces arbitrary
slowly varying factors to analytic ones and proves Theorem~\ref{thm:main} and Corollary~\ref{thm:pure}, assuming
Theorem~\ref{thm:engine}. Sections~\ref{sec:frame}--\ref{sec:extraction} prove Theorem~\ref{thm:engine}: the common
frame (Section~\ref{sec:frame}), the outer scalar function in the three geometries $p>1$, $p=1$ and $p<1$
(Section~\ref{sec:outer}), the comparison at the hard edge (Section~\ref{sec:hardedge}), and the extraction of $a_n$ and
$b_n$ (Section~\ref{sec:extraction}). Section~\ref{sec:numerics} illustrates the results numerically, and
Section~\ref{sec:further} collects remarks and open problems. The four appendices contain the longer estimates of the
Riemann--Hilbert half: the construction of the outer scalar for $p\ne1$ (Appendix~\ref{app:outer}), the a priori bound
at the hard edge (Appendix~\ref{app:apriori}), two $L^2$ bounds on a star of arcs (Appendix~\ref{app:L2}), and the
uniform invertibility of the comparison problem (Appendix~\ref{app:resolvent}). A reader who is mainly interested in the
Riemann--Hilbert analysis can start at Section~\ref{sec:frame}, with the statement of Theorem~\ref{thm:engine} in hand.

Throughout, $p>0$ is fixed, $C,C'$ denote positive constants that may change from line to line (never $c$, which is
reserved for $c(p)$; the only other constants written with $c$ are $c_0$ in Definition~\ref{def:A} and $c_1$ in
\eqref{eq:paper1-averaged}), and $n$ is the degree. We write ``Section~5'' for sections and ``\S5.3'' for
subsections.
